{{- /* Format bytes (numeric arg) as human-readable string. Usage: {{ partial "humanizeSize.html" .size }} */ -}} {{- $b := . -}} {{- if not $b -}} {{- else if lt $b 1024 -}}{{ printf "%d B" $b }} {{- else if lt $b 1048576 -}}{{ printf "%.1f KB" (div (float $b) 1024.0) }} {{- else if lt $b 1073741824 -}}{{ printf "%.1f MB" (div (float $b) 1048576.0) }} {{- else -}}{{ printf "%.2f GB" (div (float $b) 1073741824.0) }} {{- end -}}
